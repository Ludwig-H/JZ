% ---------------------------------------------------------------------
% Mise en page des notes d'orateur (document de travail, à tenir en main)
% ---------------------------------------------------------------------
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{enumitem}
\usepackage{needspace}

\definecolor{inriarouge}{HTML}{C9191E}
\definecolor{inriagris}{HTML}{384257}

% Quelques caractères que l'encodage T1 ne couvre pas tel quel
\DeclareUnicodeCharacter{2248}{\ensuremath{\approx}}
\DeclareUnicodeCharacter{021B}{\c{t}}
\DeclareUnicodeCharacter{0103}{\u{a}}

% Listes resserrées
\setlist[itemize]{leftmargin=1.25em,itemsep=2pt,topsep=3pt,parsep=0pt}
\setlist[enumerate]{leftmargin=1.7em,itemsep=2pt,topsep=3pt,parsep=0pt}

% Titre du document : compact, aligné à gauche, filet rouge Inria
\makeatletter
\renewcommand{\maketitle}{%
  \begingroup\raggedright
  {\LARGE\bfseries\color{inriagris}\@title\par}%
  \vspace{0.7ex}{\color{inriarouge}\rule{\linewidth}{1.1pt}}%
  \vspace{1.4ex}\par
  \endgroup}
\makeatother

% Sections (## du markdown)
\titleformat{\section}
  {\needspace{3\baselineskip}\normalfont\large\bfseries\color{inriagris}}{}{0pt}{}
  [\vspace{-0.6ex}{\color{inriarouge}\rule{\linewidth}{0.7pt}}]
\titlespacing*{\section}{0pt}{2.6ex plus 1.2ex}{1.1ex}

% Entrées planche par planche (### du markdown) : on évite qu'une entrée
% soit coupée par un saut de page.
\titleformat{\subsection}
  {\needspace{4.2\baselineskip}\normalfont\bfseries\color{inriarouge}}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{2.3ex plus 0.7ex minus 0.3ex}{0.45ex}

\titleformat{\subsubsection}{\normalfont\bfseries\color{inriagris}}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{1.6ex}{0.4ex}

% Étiquettes « À dire. », « Transition. », « En réserve. »
\newcommand{\notelabel}[1]{\textcolor{inriarouge}{\textbf{#1}}}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\footnotesize\color{inriagris}\runninghead}
\fancyhead[R]{\footnotesize\color{inriagris}\thepage}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[R]{\footnotesize\color{inriagris}\thepage}}

% Paragraphes resserrés : une étiquette de conduite par ligne, sans trou
\setlength{\parskip}{3.2pt plus 1pt minus 0.6pt}

\widowpenalty=10000
\clubpenalty=10000
\raggedbottom
