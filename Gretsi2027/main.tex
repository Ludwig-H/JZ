\section{Introduction}
% ============================================================
\frame{\sectionpage}

\begin{frame}{De l'image à la carte thématique}
  \begin{itemize}
    \item Associer à \hl{chaque pixel} une étiquette d'un ensemble prédéfini de classes
          sémantiques\refc{duda1973pattern}\refc{swain1978remote}\refc{richards1986remote} :
          problème étudié depuis des décennies sous les noms de \emph{classification
          supervisée}, \emph{pixel à pixel}, \emph{contextuelle}
    \item Produit une \hl{carte dense}, à distinguer de la segmentation « bas niveau »,
          qui partitionne en régions homogènes \emph{sans} leur associer de signification
    \item En télédétection : transformer les observations satellitaires en
          \hl{cartes d'occupation et d'utilisation du sol}
  \end{itemize}
  \vspace{0.4ex}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}\centering
      \includegraphics[height=1.6cm]{imgs/article/fig1a_zeebruges_image_aerienne}
      \legende{(a) image aérienne}
    \end{column}
    \begin{column}{0.49\textwidth}\centering
      \includegraphics[height=1.6cm]{imgs/article/fig1b_zeebruges_carte_segmentation}
      \legende{(b) carte de segmentation}
    \end{column}
  \end{columns}
  \credit{Jeu de données \emph{Zeebruges}, comité technique IADF de l'IEEE GRSS ;
    images et vérité terrain : Académie royale militaire belge et ONERA.}
  \biblio{duda1973pattern,swain1978remote,richards1986remote}
\end{frame}

\begin{frame}{Ce qui distingue la télédétection}
  \begin{itemize}
    \item \hl{Capteurs} hétérogènes : optique, radar à synthèse d'ouverture (RSO),
          LiDAR
    \item \hl{Résolutions} spatiales très variables, acquisitions à
          \hl{différentes dates}, \hl{plusieurs modalités} combinées
    \item Scènes géographiques complexes, dont les propriétés varient selon les
          régions, les saisons et les conditions d'acquisition
    \item Conséquence : l'évolution du domaine \hl{n'est pas} une simple adaptation
          des avancées de la vision par ordinateur
    \item C'est une convergence entre reconnaissance statistique des formes,
          traitement du signal et des images, modélisation probabiliste,
          morphologie mathématique, géométrie stochastique, analyse orientée objet
          et, récemment, apprentissage profond
  \end{itemize}
\end{frame}

\begin{frame}{Formalisation et fil directeur}
  \formulebox{$I : \Omega \longrightarrow \mathbb{R}^{d}
              \qquad\text{puis}\qquad
              f : \Omega \longrightarrow \mathcal{C}$}
  \begin{itemize}
    \item $\Omega$ : domaine spatial ; $d$ : nombre de variables (bande
          panchromatique, bandes multi- ou hyperspectrales, polarisations RSO,
          mesures LiDAR, indices spectraux, modèles numériques de terrain\dots)
    \item $\mathcal{C}$ : ensemble fini des classes sémantiques
    \item \hl{Fil directeur de l'exposé} : une intégration progressive de niveaux
          croissants de contexte
  \end{itemize}
  \vspace{0.4ex}
  \filrouge
\end{frame}

\section{Classification statistique pixel à pixel}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Les premières approches (années 1970)}
  \begin{itemize}
    \item Landsat (optique) et SeaSat (radar) créent le besoin de cartes
          d'occupation du sol automatiques\refc{schowengerdt2007remote}\refc{landgrebe2003signal}\refc{ulaby2015microwave}
    \item Chaque pixel est un vecteur d'observations
          $\mathbf{x} = (x_1,\dots,x_d)^{\mathsf{T}}$ : réponses radiométriques,
          intensités RSO, indices, données auxiliaires\refc{duda1973pattern}
  \end{itemize}
  \formulebox{$\hat{\omega} \;=\; \arg\max_{\omega_i}\; P(\omega_i \mid \mathbf{x})$}
  \begin{itemize}
    \item Sous hypothèse gaussienne multivariée : \hl{classifieur MAP}, dit informellement
          « maximum de vraisemblance » (MLC), référence pendant plusieurs
          décennies\refc{swain1978remote}\refc{richards1986remote}\refc{mather2004computer}
  \end{itemize}
  \biblio{duda1973pattern,swain1978remote,richards1986remote,schowengerdt2007remote,landgrebe2003signal,ulaby2015microwave,mather2004computer}
\end{frame}

\begin{frame}{Deux limites structurelles}
  \begin{itemize}
    \item \hl{Malédiction de la dimension} (phénomène de Hughes, décrit dès
          1968)\refc{hughes1968mean}
      \begin{itemize}
        \item plus de variables spectrales $\neq$ meilleures performances lorsque
              le nombre d'échantillons d'apprentissage reste limité
        \item les signatures spectrales seules ne suffisent pas
      \end{itemize}
    \item \hl{Hypothèse d'indépendance des observations}
      \begin{itemize}
        \item chaque pixel est décidé isolément, sans exploiter son voisinage
        \item or les objets géographiques sont fortement corrélés spatialement :
              parcelle agricole, forêt, zone urbaine, réseau routier
      \end{itemize}
    \item Conséquence : cartes bruitées, effet \hl{« poivre et sel »}, d'autant plus
          marqué que la résolution spatiale augmente
  \end{itemize}
  \biblio{hughes1968mean}
\end{frame}

\begin{frame}{L'effet « poivre et sel »}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}\centering
      \includegraphics[width=0.72\textwidth]{imgs/article/fig2a_ikonos_fausses_couleurs}
      \legende{(a) image IKONOS, 4~m, fausses couleurs}
    \end{column}
    \begin{column}{0.49\textwidth}\centering
      \includegraphics[width=0.72\textwidth]{imgs/article/fig2b_classification_pixel_a_pixel}
      \legende{(b) classification pixel à pixel}
    \end{column}
  \end{columns}
  \credit{Image IKONOS, 4~m, 3 bandes, fausses couleurs (infrarouge voisin, rouge,
    bleu) ; carte de segmentation \cite{moser2013landcover}.}
  \vspace{0.6ex}
  \begin{itemize}
    \item La décision ne dépend plus seulement de la radiométrie d'un pixel :
          il faut son \hl{environnement spatial}
    \item Le problème cesse d'être une classification spectrale pour devenir un
          problème d'\hl{interprétation spatiale des
          scènes}\refc{geman1984stochastic}\refc{besag1986statistical}\refc{solberg1996markov}
  \end{itemize}
  \biblio{geman1984stochastic,besag1986statistical,solberg1996markov,moser2013landcover}
\end{frame}

\section{Méthodes discriminantes et à noyaux}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Estimer la frontière plutôt que la distribution}
  \begin{itemize}
    \item À partir de la fin des années 1990 : estimer \hl{directement les
          frontières de décision} plutôt que la loi des observations
    \item Réponse à deux difficultés : hypothèses gaussiennes peu réalistes,
          dimension élevée des données
    \item \hl{Machines à vecteurs de support} (SVM), issues de la théorie de
          l'apprentissage statistique\refc{vapnik1998statistical}\refc{cortes1995support}
      \begin{itemize}
        \item principe de \hl{maximisation de la marge} entre classes
        \item excellente généralisation même lorsque le nombre d'échantillons
              d'apprentissage est faible devant la dimension
      \end{itemize}
  \end{itemize}
  \biblio{vapnik1998statistical,cortes1995support}
\end{frame}

\begin{frame}{Noyaux : la non-linéarité sans perdre la convexité}
  \begin{itemize}
    \item Les méthodes à noyaux étendent le cadre aux problèmes non linéaires
          tout en conservant une \hl{formulation convexe}\refc{scholkopf2002learning}
    \item Années 2000 : les SVM deviennent l'un des classifieurs les plus utilisés
          en télédétection : multispectral, hyperspectral, multisources
    \item Développement de noyaux adaptés aux observations
          géospatiales\refc{campsvalls2009kernel}\refc{mountrakis2011support}
    \item \hl{Mais} : les SVM améliorent la \emph{fonction de décision}, pas la
          représentation : appliqués aux vecteurs d'observations, ils restent
          des classifieurs \hl{pixel à pixel}
    \item Deux directions de recherche évoluent alors en parallèle :
          \emph{nouveaux classifieurs} et \emph{représentations spatiales plus riches}
  \end{itemize}
  \biblio{scholkopf2002learning,campsvalls2009kernel,mountrakis2011support}
\end{frame}

\section{Méthodes d'ensemble}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Forêts aléatoires}
  \begin{itemize}
    \item Combiner plusieurs classifieurs pour gagner en robustesse et en
          généralisation : \hl{apprentissage d'ensemble}
    \item \hl{Forêts aléatoires} : arbres de décision appris sur des sous-ensembles
          aléatoires \emph{des observations} et \emph{des variables}, agrégés par
          vote majoritaire\refc{breiman2001random}
    \item Limitent le surapprentissage tout en modélisant des relations non linéaires
    \item En pratique : peu de réglages, robustes au bruit, supportent un grand
          nombre de variables, fournissent des mesures d'\hl{importance des variables}
    \item Multispectral et hyperspectral\refc{gislason2006random}, puis occupation du
          sol, détection de changements, fusion multisources\refc{belgiu2016random}
  \end{itemize}
  \biblio{breiman2001random,gislason2006random,belgiu2016random}
\end{frame}

\begin{frame}{Bilan de l'ère « pixel »}
  \begin{itemize}
    \item Méthodes discriminantes et d'ensemble : gains nets \hl{sur le classifieur}
    \item Elles restent \hl{indépendantes de la représentation}, qu'elle soit
          spectrale, texturale, morphologique ou spectro-spatiale
    \item La limite fondamentale demeure : \hl{l'organisation spatiale de la scène
          n'est pas prise en compte explicitement}
    \item Deux réponses complémentaires vont structurer la suite :
      \begin{itemize}
        \item régulariser les \hl{étiquettes} : contextuel, markovien,
              variationnel (§5, §6, §8)
        \item enrichir les \hl{descripteurs} : spectro-spatial, orienté objet (§7)
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Modèles markoviens et bayésiens}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Les champs de Markov : la première formalisation}
  \begin{itemize}
    \item Les étiquettes ne sont plus indépendantes : ce sont des variables
          aléatoires définies sur un \hl{graphe de
          voisinage}\refc{geman1984stochastic}\refc{besag1986statistical}\refc{kato2012markov}
    \item Hypothèses de Markov \emph{et} équivalence champs de Markov /
          distributions de Gibbs
  \end{itemize}
  \formulebox{$\hat{Y} \;=\; \arg\max_{Y} P(Y \mid X)
              \qquad\Longleftrightarrow\qquad
              \hat{Y} \;=\; \arg\min_{Y} U(Y \mid X)$}
  \begin{itemize}
    \item $X$ : les observations, $Y$ : les étiquettes à estimer
    \item Le théorème de Bayes transforme l'estimation MAP en une
          \hl{minimisation d'énergie}
  \end{itemize}
  \biblio{geman1984stochastic,besag1986statistical,kato2012markov}
\end{frame}

\begin{frame}{Attache aux données + régularisation}
  \formulebox{$U(Y \mid X) \;=\; U_d(Y \mid X) \;+\; \beta\, U_r(Y \mid X)$}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \item $U_d$ : \hl{attache aux données}, dérivée de la vraisemblance
        \item $U_r$ : \hl{régularisation spatiale}, favorise la cohérence entre
              pixels voisins
        \item $\beta$ : compromis fidélité / régularité
        \item Voisinages du 1\textsuperscript{er} ou du 2\textsuperscript{d} ordre ;
              les potentiels de clique pénalisent les changements de classe entre
              pixels adjacents\refc{kato2012markov}
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}\centering
      \includegraphics[height=2.5cm]{imgs/article/fig3a_voisinage_ordre1_cliques}%
      \hspace{2mm}%
      \includegraphics[height=2.5cm]{imgs/article/fig3b_voisinage_ordre2_cliques}
      \legende{(a) 1\textsuperscript{er} ordre \quad (b) 2\textsuperscript{d} ordre}
      \credit{Voisinages entre pixels dans une grille et cliques
        associées \cite{kato2012markov}.}
    \end{column}
  \end{columns}
  \biblio{kato2012markov}
\end{frame}

\begin{frame}{Optimiser : un problème difficile}
  \begin{itemize}
    \item Trouver la configuration optimale est un problème d'optimisation
          combinatoire difficile $\Rightarrow$ stratégies \hl{sous-optimales}
    \item Modes conditionnels itérés (\hl{ICM}), \hl{recuit simulé}\refc{kirkpatrick1984optimization},
          échantillonneur de \hl{Gibbs}\refc{geman1984stochastic}
    \item \hl{Coupes de graphes}\refc{boykov2001fast}, \hl{propagation de croyance avec
          boucles}\refc{tanaka2003loopy}
    \item Domaines d'application : cohérence spatiale des classifications,
          segmentation multisource, restauration, détection de changements, fusion
          optique / radar / LiDAR\refc{solberg1996markov}\refc{derin1987modeling}\refc{moser2013combining}
    \item Au-delà des applications : une \hl{formulation probabiliste générale de la
          régularisation spatiale}
  \end{itemize}
  \biblio{geman1984stochastic,solberg1996markov,kirkpatrick1984optimization,boykov2001fast,tanaka2003loopy,derin1987modeling,moser2013combining}
\end{frame}

\begin{frame}{Des MRF aux CRF}
  \begin{itemize}
    \item Les champs conditionnels aléatoires représentent directement $P(Y\mid X)$ plutôt
          que la loi jointe\refc{lafferty2001conditional} : ils facilitent l'usage de
          \hl{caractéristiques discriminantes complexes}
    \item \hl{CRF entièrement connectés} : contours des objets mieux
          préservés\refc{krahenbuhl2011efficient}
    \item Intégration aux réseaux profonds : modules de raffinement, \hl{couches
          différentiables}\refc{zheng2015conditional} ; cadres markoviens hiérarchiques et
          CRF appris\refc{pastorino2021multisensor}\refc{pastorino2022semantic}\refc{pastorino2024crfnet}\refc{voisin2013classification}
    \item \hl{L'idée qui traverse le domaine} : une carte doit être compatible avec les
          observations locales \emph{et} avec l'organisation spatiale de la scène
  \end{itemize}
  \biblio{lafferty2001conditional,krahenbuhl2011efficient,zheng2015conditional,pastorino2021multisensor,pastorino2022semantic,pastorino2024crfnet,voisin2013classification}
\end{frame}

\section{Méthodes spectro-spatiales}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Enrichir la représentation, pas les étiquettes}
  \begin{itemize}
    \item Les modèles contextuels régularisent les \hl{étiquettes} ; ici on enrichit
          les \hl{descripteurs} \emph{avant} la classification\refc{landgrebe2003signal}
    \item Idées : montée en résolution spatiale et spectrale des capteurs
      \begin{itemize}
        \item une même classe présente une forte variabilité spectrale
        \item des objets différents ont parfois des signatures très proches
      \end{itemize}
    \item Texture, forme, taille, organisation spatiale et relations de voisinage
          deviennent \hl{aussi discriminantes que les valeurs
          spectrales}\refc{dellepiane1991sar}
  \end{itemize}
  \biblio{landgrebe2003signal,dellepiane1991sar}
\end{frame}

\begin{frame}{Profils morphologiques et profils d'attributs}
  \begin{itemize}
    \item \hl{Profils morphologiques étendus} (EMP)\refc{benediktsson2005classification} puis

          les propriétés spectrales et la structure spatiale \hl{à plusieurs échelles}

    \item Utiliser les \hl{régions segmentées} comme unités d'analyse plutôt que les

  \end{itemize}
  \biblio{campsvalls2009kernel,benediktsson2005classification,dallamura2010morphological,fauvel2008spectral,plaza2009recent,campsvalls2014advances,tarabalka2009spectral}
\end{frame}

\section{Méthodes variationnelles}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Segmenter en minimisant une énergie}
  \begin{itemize}
    \item Fin des années 1980, en parallèle des champs de Markov : \hl{pas de loi de
          probabilité explicite sur les étiquettes}, mais la recherche de la partition
          qui réalise le meilleur compromis entre fidélité aux observations et
          régularité\refc{kato2012markov}\refc{mumford1989optimal}\refc{samson2000variational}
    \item \hl{Contours actifs} : les frontières sont des courbes déformables évoluant
          sous l'action\refc{samson2000variational}\refc{kass1988snakes}
      \begin{itemize}
        \item de \emph{forces internes} (régularité du contour)
        \item de \emph{forces externes} (dérivées de l'information de l'image)
      \end{itemize}
    \item Une représentation \hl{explicite} des contours, et le point de départ d'une
          large famille de méthodes par minimisation d'énergie
  \end{itemize}
  \biblio{kato2012markov,mumford1989optimal,samson2000variational,kass1988snakes}
\end{frame}

\begin{frame}{La fonctionnelle de Mumford--Shah}
  \formulebox[0.96]{$\displaystyle E(u,K) \;=\;
     \underbrace{\int_{\Omega} (u - I)^2\,\mathrm{d}x}
                _{\textcolor{inria-rouge}{\text{\scriptsize fidélité aux données}}}
     \;+\; \mu \underbrace{\int_{\Omega \setminus K} |\nabla u|^2\,\mathrm{d}x}
                _{\textcolor{inria-rouge}{\text{\scriptsize régularité intra-région}}}
     \;+\; \nu\, \underbrace{|K|\vphantom{\int}}
                _{\textcolor{inria-rouge}{\text{\scriptsize complexité des contours}}}$}
  \begin{itemize}
    \item $u$ : image reconstruite, $I$ : image observée, $K$ : ensemble des
          discontinuités (les contours)\refc{mumford1989optimal}
    \item $\mu > 0$ : douceur de la solution à l'intérieur des régions ;
          $\nu > 0$ : pondération de la mesure des contours
    \item Un compromis explicite entre \hl{ajustement aux données} et
          \hl{régularité de la solution}, principe commun à de nombreuses méthodes
          de segmentation
    \item L'un des modèles variationnels les plus influents en traitement d'images
  \end{itemize}
  \biblio{mumford1989optimal}
\end{frame}

\begin{frame}{Robustesse numérique et applications}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      \begin{itemize}
        \item \hl{Surfaces de niveau} : contours représentés implicitement, changements
              de topologie gérés automatiquement\refc{osher1988fronts}
        \item \hl{Modèles régionaux} : statistiques des régions plutôt que gradients
              seuls, robustes au bruit et au faible
              contraste\refc{chan2001active}
        \item En télédétection : segmentation optique haute résolution, restauration,
              fusion, extraction d'objets ; classification et restauration unifiées
              dans une même fonctionnelle\refc{samson2000variational}
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}\centering
      \includegraphics[width=0.485\textwidth]{imgs/article/fig5a_spot_cnes}%
      \hspace{1mm}%
      \includegraphics[width=0.485\textwidth]{imgs/article/fig5b_segmentation_variationnelle}
      \legende{(a) image SPOT \quad (b) segmentation}
      \credit{(a) image satellitaire SPOT \copyright{} CNES ; (b) segmentation par la
        méthode variationnelle de \cite{samson2000variational}.}
    \end{column}
  \end{columns}
  \vspace{0.6ex}
  \begin{itemize}
    \item \hl{Convergence} : bayésien, champs de Markov, variationnel et formulations
          sur graphes sont différentes expressions d'un même principe :
          \hl{attache aux données $+$ régularisation}
  \end{itemize}
  \biblio{samson2000variational,osher1988fronts,chan2001active}
\end{frame}

\section{Apprentissage profond}
% ============================================================
\frame{\sectionpage}

\begin{frame}{2012 : la bascule}
  \begin{itemize}
    \item \hl{AlexNet} démontre une amélioration spectaculaire en classification
          d'images naturelles\refc{krizhevsky2012imagenet}
    \item On \hl{apprend} des représentations hiérarchiques directement à partir des
          données, sans conception manuelle de descripteurs spectraux, texturaux ou
          géométriques
    \item La performance dépend désormais de la capacité du réseau à apprendre
          \hl{conjointement} une représentation pertinente \emph{et} la fonction de
          décision
    \item Transposition rapide à la télédétection : représentations plus
          discriminantes, \hl{transfert} de modèles pré-entraînés sur les grandes bases
          d'images naturelles\refc{chen2014deep}\refc{castelluccio2015land}
    \item Les approches spectro-spatiales passent de représentations \emph{conçues} à
          des représentations \emph{apprises}
  \end{itemize}
  \biblio{krizhevsky2012imagenet,chen2014deep,castelluccio2015land}
\end{frame}

\begin{frame}{Prédiction dense : FCN et encodeur--décodeur}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \hl{FCN} : suppression des couches entièrement connectées au profit
              d'opérations convolutionnelles $\Rightarrow$ prédiction dense à l'échelle
              du pixel, apprise de bout en bout\refc{long2015fully}
        \item \hl{U-Net} : encodeur--décodeur dont les \emph{connexions de saut}
              réinjectent la localisation perdue au sous-échantillonnage, d'où de
              meilleurs contours et de plus petits objets\refc{ronneberger2015unet}
        \item Applications : occupation du sol, bâtiments, routes, réseaux
              hydrographiques\refc{audebert2018beyond}\refc{zhang2018road}\refc{isikdogan2017surface}
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}\centering
      \vspace{0.4cm}
      \includegraphics[width=\textwidth]{imgs/article/fig6_fcn_potsdam}
      \legende{architecture d'un FCN}
      \credit{Appliquée aux images aériennes de la base ISPRS \emph{2D Semantic
        Labeling Challenge} (Potsdam) \cite{pastorino2026probabilistic}.}
    \end{column}
  \end{columns}
  \biblio{long2015fully,pastorino2026probabilistic,ronneberger2015unet,audebert2018beyond,zhang2018road,isikdogan2017surface}
\end{frame}

\begin{frame}{Contexte, échelles, modalités}
  \begin{itemize}
    \item \hl{Réseaux résiduels} : apprendre des modèles plus profonds en limitant la
          disparition du gradient\refc{he2016deep}
    \item \hl{DeepLab} : convolutions dilatées et agrégation multi-échelle (ASPP)
          pour augmenter le champ réceptif \emph{sans} dégrader la résolution des
          prédictions\refc{chen2017rethinking}\refc{chen2018encoder}
    \item \hl{Pyramides de caractéristiques} : objets de tailles très
          variables\refc{lin2017feature}, décisif en télédétection où une même scène
          contient véhicules, bâtiments, parcelles et massifs forestiers
    \item \hl{Multimodalité} : apprendre des représentations communes à partir de
          l'optique, du RSO, du LiDAR, des modèles numériques de terrain et de données
          géographiques auxiliaires\refc{audebert2018beyond}\refc{ma2019deep}
  \end{itemize}
  \biblio{audebert2018beyond,he2016deep,chen2017rethinking,chen2018encoder,lin2017feature,ma2019deep}
\end{frame}

\begin{frame}{Ce que l'apprentissage profond ne supprime pas}
  \begin{itemize}
    \item Contexte spatial, représentation multi-échelle, fusion de données,
          hiérarchie des objets : \hl{reformulés} dans un cadre d'apprentissage de bout
          en bout, non abandonnés
    \item Limites persistantes des réseaux convolutifs :
      \begin{itemize}
        \item annotations pixel à pixel \hl{coûteuses}, particulièrement en télédétection
        \item \hl{généralisation} limitée entre capteurs, régions géographiques et
              modalités d'acquisition
        \item difficulté à modéliser les dépendances spatiales à \hl{très longue portée}
      \end{itemize}
    \item Ces limites ouvrent la voie aux mécanismes d'attention, puis aux modèles de
          fondation géospatiaux
  \end{itemize}
\end{frame}

\section{Transformers et modèles de fondation}
% ============================================================
\frame{\sectionpage}

\begin{frame}{L'attention en segmentation}
  \begin{itemize}
    \item Développés pour le traitement du langage, les « transformers » modélisent
          directement les relations entre \hl{éléments distants} d'une
          séquence\refc{vaswani2017attention} ; leur adaptation à la vision ouvre une
          nouvelle étape\refc{dosovitskiy2021image}
    \item \hl{Swin Transformer}\refc{liu2021swin} et \hl{SegFormer}\refc{xie2021segformer} :
          combiner une représentation \emph{hiérarchique} de l'image et une
          modélisation \emph{globale} du contexte
    \item Particulièrement pertinent en télédétection : l'identification d'un bâtiment,
          d'une route ou d'une parcelle ne repose pas seulement sur l'apparence locale,
          mais sur son \hl{organisation au sein de la scène}
  \end{itemize}
  \biblio{vaswani2017attention,dosovitskiy2021image,liu2021swin,xie2021segformer}
\end{frame}

\begin{frame}{Apprendre sans annotations}
  \begin{itemize}
    \item L'\hl{apprentissage auto-supervisé} devient un axe majeur de recherche
    \item Les \hl{autoencodeurs masqués} montrent qu'il est possible d'apprendre des
          représentations visuelles riches à partir de grandes quantités de données
          \hl{non annotées}\refc{he2022masked}
    \item Réponse directe à un déséquilibre propre au domaine :
      \begin{itemize}
        \item les archives satellitaires sont \hl{abondantes}
        \item les annotations pixel à pixel restent \hl{rares et coûteuses}
      \end{itemize}
  \end{itemize}
  \biblio{he2022masked}
\end{frame}

\begin{frame}{Modèles de fondation géospatiaux}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Pré-entraînés sur de très grands volumes, puis adaptés : classification,
              segmentation, détection d'objets et de
              changements\refc{brown2025alphaearth}\refc{cong2022satmae}\refc{jakubik2023foundation}\refc{wang2023empirical}\refc{kirillov2023segment}
        \item Objectif : des représentations \hl{génériques}, transférables entre
              capteurs, régions géographiques et applications
        \item \hl{Coût} du pré-entraînement (calcul, énergie) $\Rightarrow$ empreinte
              environnementale et accessibilité
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}\centering
      \includegraphics[width=\textwidth]{imgs/article/fig7_modele_fondation_geospatial}
      \legende{principe général d'un modèle de fondation géospatial}
    \end{column}
  \end{columns}
  \vspace{0.4ex}
  \begin{itemize}
    \item Pas une rupture : les mêmes questions (hétérogénéité, contexte, généralisation)
          sont désormais \hl{apprises} plutôt que \hl{construites}
  \end{itemize}
  \biblio{brown2025alphaearth,cong2022satmae,jakubik2023foundation,wang2023empirical,kirillov2023segment}
\end{frame}

\section{Conclusion}
% ============================================================
\frame{\sectionpage}

\begin{frame}{Une histoire non linéaire}
  \begin{itemize}
    \item Pas une trajectoire allant simplement des classifieurs statistiques aux
          réseaux profonds, mais la \hl{convergence de plusieurs traditions} :
          classification statistique, méthodes à noyaux, méthodes d'ensemble, modèles
          contextuels, champs de Markov, méthodes variationnelles, géométrie
          stochastique, apprentissage profond, modèles de fondation
    \item Une lecture unifiée : l'\hl{intégration progressive de niveaux croissants de
          contexte}
  \end{itemize}
  \vspace{0.6ex}
  \filrouge
  \vspace{0.4ex}
  \begin{itemize}
    \item Un principe récurrent : \hl{attache aux données $+$ régularisation},
          des champs de Markov aux fonctions de perte des réseaux profonds
  \end{itemize}
\end{frame}

\begin{frame}{Perspectives}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Généralisation géographique
        \item Manque d'annotations
        \item Fusion optique / radar / LiDAR
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Prise en compte de la temporalité
        \item Interprétabilité des modèles
        \item Robustesse opérationnelle
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1.2ex}
  \begin{itemize}
    \item La segmentation sémantique en télédétection demeure un \hl{domaine ouvert},
          à l'intersection du traitement du signal et des images, de la modélisation
          probabiliste, de la géométrie, de la vision par ordinateur et de
          l'intelligence artificielle, dans un contexte géospatial
  \end{itemize}
\end{frame}
